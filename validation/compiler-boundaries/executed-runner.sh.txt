#!/usr/bin/env bash
set -uo pipefail
base=/workspace/shared/compiler-boundary-probes
src=$base/source
out=$base/results
mkdir -p "$out"
cd "$src"
printf 'Pinned source: '; git -C /workspace/shared/forth-bootstrap-book-review/forth-bootstrap rev-parse HEAD
printf 'Private tmp type: '; findmnt -n -o FSTYPE /tmp | head -1
# Chapter 37 core profile and its own source-buffer/ELF hook.
cat forth/010-lib.fth cc/020-cc-arena.fth cc/030-cc-io.fth \
 cc/040-cc-prep.fth cc/050-cc-lex.fth cc/060-cc-types.fth cc/070-cc-sym.fth \
 cc/080-cc-elf.fth cc/090-cc-emit.fth cc/100-cc-expr.fth \
 cc/110-cc-decl.fth cc/112-cc-stmt.fth cc/114-cc-func.fth cc/116-cc-prog.fth > "$out/core.fth"
# Chapter 39 native profile: full ordered loader, definitions without core invocation.
layers=$(cc/compiler-layers.sh)
cat forth/010-lib.fth $layers | sed '$d' > "$out/native.fth"
cat >> "$out/native.fth" <<'FORTH'
true cc-target-lp64 !
: storage-main
  cc-load-stdin cc-preprocess cc-out-init cc-globals-init
  cc-emit-elf-header cc-native-program cc-finalize-globals
  cc-finalize-elf cc-out-path cc-write-output bye ;
storage-main
FORTH
# Exact Chapter 47 object-layer list; deliberately excludes later 129/131/132.
cat forth/010-lib.fth cc/0[2-9]*-cc-*.fth cc/1[01]*-cc-*.fth \
 cc/121-cc-sysv.fth cc/123-cc-object-program.fth cc/124-cc-target.fth \
 cc/125-cc-consteval.fth cc/126-cc-varargs.fth cc/127-cc-binary64.fth \
 cc/128-cc-float-literal.fth > "$out/object47.fth"
# Chapter 47's small _start object.
{
 cat forth/010-lib.fth cc/020-cc-arena.fth cc/030-cc-io.fth cc/081-cc-object.fth
 printf 'create output-path s, %s/start.o [lit] 0 c,\n' "$out"
 cat <<'FORTH'
cc-obj-init
[lit] 232 cc-obj-byte [lit] 0 cc-obj-4le
[lit] 137 cc-obj-byte [lit] 199 cc-obj-byte
[lit] 184 cc-obj-byte [lit] 60 cc-obj-4le
[lit] 15 cc-obj-byte [lit] 5 cc-obj-byte
create entry-name s, _start
create main-name s, main
entry-name [lit] 6 cc-obj-global cc-obj-func cc-obj-default cc-obj-text [lit] 0 [lit] 14 cc-obj-symbol drop
main-name [lit] 4 cc-obj-global cc-obj-func cc-obj-default cc-obj-undef [lit] 0 [lit] 0 cc-obj-symbol
>r cc-obj-text [lit] 1 cc-obj-plt32 r> [lit] 0 [lit] 4 - cc-obj-reloc
output-path cc-obj-write bye
FORTH
} > "$out/start.fth"
status=0; timeout --kill-after=1s 10s ./seed-forth < "$out/start.fth" > "$out/start.stdout" 2> "$out/start.stderr" || status=$?
printf '%s\n' "$status" > "$out/start.status"
[ "$status" -eq 0 ] || exit "$status"
core_probe() {
 local name=$1 code=$2 status=0 runstatus=0
 local d="$out/$name"; mkdir -p "$d"; printf '%s' "$code" > "$d/probe.c"
 {
  cat "$out/core.fth"
  printf 'cc-src-init\nhere dup cc-src-buffer !\ns, %s\nhere swap - cc-src-len !\n' "$code"
  cat <<'FORTH'
[lit] 0 cc-src-pos ! [lit] 0 cc-tok-pending !
cc-out-init [lit] 0 cc-bss-size ! cc-emit-elf-header
cc-parse-program cc-finalize-globals cc-finalize-elf
[lit] 1 cc-out-buf cc-out-pos @ write drop
bye
FORTH
 } > "$d/input.fth"
 timeout --kill-after=1s 3s ./seed-forth < "$d/input.fth" > "$d/program" 2> "$d/compiler.stderr" || status=$?
 printf '%s\n' "$status" > "$d/compiler.status"
 printf '%s: compiler=%s bytes=%s' "$name" "$status" "$(wc -c < "$d/program")"
 if [ "$status" -eq 0 ]; then
  chmod +x "$d/program"
  timeout --kill-after=1s 3s "$d/program" > "$d/program.stdout" 2> "$d/program.stderr" || runstatus=$?
  printf '%s\n' "$runstatus" > "$d/program.status"
  printf ' program=%s' "$runstatus"
 fi
 printf '\n'; cat "$d/compiler.stderr"
}
native_probe() {
 local name=$1 code=$2 status=0 runstatus=0
 local d="$out/$name"; mkdir -p "$d"; printf '%s\n' "$code" > "$d/probe.c"
 cat "$out/native.fth" "$d/probe.c" > "$d/input.fth"
 timeout --kill-after=1s 10s ./seed-forth < "$d/input.fth" > "$d/compiler.stdout" 2> "$d/compiler.stderr" || status=$?
 printf '%s\n' "$status" > "$d/compiler.status"
 printf '%s: compiler=%s' "$name" "$status"
 if [ "$status" -eq 0 ] && [ -f /tmp/cc-out ]; then
  mv /tmp/cc-out "$d/program"; chmod +x "$d/program"
  timeout --kill-after=1s 3s "$d/program" > "$d/program.stdout" 2> "$d/program.stderr" || runstatus=$?
  printf '%s\n' "$runstatus" > "$d/program.status"
  printf ' program=%s' "$runstatus"
 fi
 printf '\n'; cat "$d/compiler.stdout" "$d/compiler.stderr"
}
object_probe() {
 local name=$1 code=$2 status=0 linkstatus=0 runstatus=0
 local d="$out/$name"; mkdir -p "$d"; printf '%s\n' "$code" > "$d/probe.c"
 {
  cat "$out/object47.fth"
  printf 'create output-path s, %s/probe.o [lit] 0 c,\n' "$d"
  cat <<'FORTH'
cc-sysv-object-enable
: object-main
  cc-load-stdin cc-preprocess cc-out-init cc-globals-init
  cc-sysv-object-program output-path cc-obj-write bye ;
object-main
FORTH
  cat "$d/probe.c"
 } > "$d/input.fth"
 timeout --kill-after=1s 10s ./seed-forth < "$d/input.fth" > "$d/compiler.stdout" 2> "$d/compiler.stderr" || status=$?
 printf '%s\n' "$status" > "$d/compiler.status"
 printf '%s: compiler=%s' "$name" "$status"
 if [ "$status" -eq 0 ]; then
  {
   cat forth/010-lib.fth cc/020-cc-arena.fth cc/030-cc-io.fth cc/140-cc-link.fth
   printf 'create p-start.o s, %s/start.o [lit] 0 c,\n' "$out"
   for n in probe.o program; do printf 'create p-%s s, %s/%s [lit] 0 c,\n' "$n" "$d" "$n"; done
   printf '%s\n' 'create entry-name s, _start' \
    'lnk-init p-start.o lnk-add-object p-probe.o lnk-add-object' \
    'entry-name [lit] 6 lnk-entry p-program lnk-link lnk-release bye'
  } > "$d/link.fth"
  timeout --kill-after=1s 10s ./seed-forth < "$d/link.fth" > "$d/link.stdout" 2> "$d/link.stderr" || linkstatus=$?
  printf '%s\n' "$linkstatus" > "$d/link.status"; printf ' link=%s' "$linkstatus"
  if [ "$linkstatus" -eq 0 ]; then
   timeout --kill-after=1s 3s "$d/program" > "$d/program.stdout" 2> "$d/program.stderr" || runstatus=$?
   printf '%s\n' "$runstatus" > "$d/program.status"; printf ' program=%s' "$runstatus"
  fi
 fi
 printf '\n'; cat "$d/compiler.stdout" "$d/compiler.stderr"
}
